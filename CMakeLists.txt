cmake_minimum_required(VERSION 3.20)
project(gai LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(GAI_NATIVE "Compile with -march=native (not portable across CPUs)" ON)
option(GAI_LTO "Link-time optimization" OFF)
option(GAI_LIBTORCH "Enable libtorch-based NN evaluator (requires libtorch)" OFF)
set(GAI_SANITIZE "" CACHE STRING "Semicolon list, e.g. address;undefined  (or thread)")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

find_package(Threads REQUIRED)
add_library(gai_engine INTERFACE)
target_include_directories(gai_engine INTERFACE ${CMAKE_SOURCE_DIR} ${CMAKE_BINARY_DIR}/generated)
target_compile_options(gai_engine INTERFACE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(gai_engine INTERFACE Threads::Threads)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_options(gai_engine INTERFACE -O3)
endif()
if(GAI_NATIVE)
  target_compile_options(gai_engine INTERFACE -march=native)
endif()
if(GAI_LTO)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()
if(GAI_SANITIZE)
  string(REPLACE ";" "," _san "${GAI_SANITIZE}")
  target_compile_options(gai_engine INTERFACE -fsanitize=${_san} -fno-omit-frame-pointer -g)
  target_link_options(gai_engine INTERFACE -fsanitize=${_san})
endif()

# ---- GAI_LIBTORCH: libtorch / TorchScript support ----
if(GAI_LIBTORCH)
  find_package(Torch REQUIRED)  # will print clear error if libtorch is missing
  if(TARGET Torch::torch)
    target_link_libraries(gai_engine INTERFACE Torch::torch)
  else()
    # pip-wheel torch ships imported target `torch` + TORCH_LIBRARIES / TORCH_CXX_FLAGS
    target_link_libraries(gai_engine INTERFACE ${TORCH_LIBRARIES})
    if(TORCH_CXX_FLAGS)
      target_compile_options(gai_engine INTERFACE ${TORCH_CXX_FLAGS})
    endif()
  endif()
  target_compile_definitions(gai_engine INTERFACE GAI_LIBTORCH=1)
endif()

# ---- game auto-discovery: every games/<name>/game.hpp is registered (namespace game_<name>) ----
file(GLOB _dirs LIST_DIRECTORIES true ${CMAKE_SOURCE_DIR}/games/*)
set(_inc "// GENERATED by CMake. Do not edit.\n")
set(_list "#define GAI_GAME_LIST(F) \\\n")
set(GAI_GAMES "")
foreach(d ${_dirs})
  if(EXISTS ${d}/game.hpp)
    get_filename_component(n ${d} NAME)
    list(APPEND GAI_GAMES ${n})
    string(APPEND _inc "#include \"${d}/game.hpp\"\n")
    if(EXISTS ${d}/reference_game.hpp)
      string(APPEND _inc "#include \"${d}/reference_game.hpp\"\n")
      string(APPEND _list "  F(${n}, game_${n}::Game, game_${n}::RefGame) \\\n")
    else()
      string(APPEND _list "  F(${n}, game_${n}::Game, void) \\\n")
    endif()
  endif()
endforeach()
string(APPEND _list "\n")
file(GENERATE OUTPUT ${CMAKE_BINARY_DIR}/generated/games.inc CONTENT "${_inc}${_list}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_dirs})

foreach(t validate_game benchmark_game selfplay evaluate play)
  add_executable(${t} tools/${t}.cpp)
  target_link_libraries(${t} PRIVATE gai_engine)
endforeach()

enable_testing()
foreach(t test_time_manager test_mcts)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} PRIVATE gai_engine)
  add_test(NAME ${t} COMMAND ${t})
endforeach()
foreach(g ${GAI_GAMES})
  add_test(NAME validate_${g} COMMAND validate_game --game ${g} --games 300)
endforeach()
message(STATUS "Games: ${GAI_GAMES}")
